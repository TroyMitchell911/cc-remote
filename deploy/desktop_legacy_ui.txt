async function serveAssets(browserSession) {
  const base = resolve(here, '../web');
  const assets = new Map();
  const walk = async (path, prefix = '') => {
    for (const entry of await readdir(path, { withFileTypes: true })) {
      if (entry.isDirectory()) await walk(join(path, entry.name), `${prefix}/${entry.name}`);
      else if (entry.isFile()) assets.set(`${prefix}/${entry.name}`, join(path, entry.name));
    }
  };
  await walk(base);
  assets.set('/', join(base, 'index.html'));
  const mime = { '.html': 'text/html; charset=utf-8', '.js': 'text/javascript', '.css': 'text/css',
    '.json': 'application/json', '.svg': 'image/svg+xml', '.png': 'image/png', '.woff2': 'font/woff2' };
  const handler = async request => {
    const url = new URL(request.url);
    const file = url.origin === origin && request.method === 'GET' && assets.get(url.pathname);
    if (file) return new Response(await readFile(file), { headers: {
      'Content-Type': mime[/\.[^.]+$/.exec(file)?.[0]] || 'application/octet-stream',
      'Cache-Control': url.pathname.startsWith('/assets/') ? 'public,max-age=31536000,immutable' : 'no-store',
    } });
    return forwardShellRequest(request, browserSession, origin);
  };
  browserSession.protocol.handle('https', handler);
  browserSession.protocol.handle('http', handler);
}
